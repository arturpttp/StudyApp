# Export (PDF + JSON) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Export exam reports and stats snapshots to PDF + JSON, and answer history to JSON, all client-side. One new read-only endpoint serves the answer history.

**Architecture:** Pure modules under `apps/web/src/lib/export/` produce Blobs or strings from typed inputs (testable). jsPDF + html2canvas handle PDF generation and chart capture. Two client-side button bundles mount on `/exams/[id]` and `/stats`. One new `GET /api/v1/history/answers` endpoint adds the only server-side surface.

**Tech Stack:** Next.js 16.2.3 App Router, Prisma 7.7, Auth.js v5 (JWT), Zod + `@asteasolutions/zod-to-openapi`, Kubb + TanStack Query, Tailwind v4, recharts ^3.x, **jspdf ^3.x + html2canvas ^1.4 (new)**, Vitest + real Postgres.

function pad2(n: number): string {
  return n < 10 ? "0" + n : String(n);
}

export function formatExportFilename(
  prefix: string,
  ext: "pdf" | "json",
  now: Date,
): string {
  const yyyy = now.getUTCFullYear();
  const mm = pad2(now.getUTCMonth() + 1);
  const dd = pad2(now.getUTCDate());
  const hh = pad2(now.getUTCHours());
  const mi = pad2(now.getUTCMinutes());
  return `${prefix}-${yyyy}${mm}${dd}-${hh}${mi}.${ext}`;
}
```

export function downloadBlob(blob: Blob, filename: string): void {
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(url);
}

import { prisma } from "@healthquest/db";
import { requireAuth } from "@/lib/api/require-auth";

export async function GET(): Promise<Response> {
  const { session, errorResponse } = await requireAuth();
  if (errorResponse) return errorResponse;

  const answers = await prisma.answerHistory.findMany({
    where: { userId: session.user.id },
    orderBy: { createdAt: "desc" },
    select: {
      id: true,
      createdAt: true,
      isCorrect: true,
      responseTime: true,
      question: {
        select: { id: true, statement: true, difficulty: true },
      },
      alternative: {
        select: { id: true, text: true, position: true },
      },
    },
  });

  return Response.json(answers);
}
```

import { jsPDF } from "jspdf";
import type { ExamReportData } from "./exam-json";

const MARGIN = 40;
const PAGE_WIDTH = 595; // A4 portrait
const PAGE_HEIGHT = 842;
const CONTENT_WIDTH = PAGE_WIDTH - MARGIN * 2;

function ensureSpace(doc: jsPDF, y: number, needed: number): number {
  if (y + needed > PAGE_HEIGHT - MARGIN) {
    doc.addPage();
    return MARGIN;
  }
  return y;
}

function letterFor(position: number): string {
  return String.fromCharCode("A".charCodeAt(0) + position);
}

export function buildExamPdf(exam: ExamReportData, now: Date): Blob {
  const doc = new jsPDF({ unit: "pt", format: "a4" });
  let y = MARGIN;

  doc.setFontSize(18);
  doc.text("Relatório de Simulado", MARGIN, y);
  y += 24;

  doc.setFontSize(10);
  doc.setTextColor(100);
  doc.text(`Gerado em ${now.toLocaleString("pt-BR")}`, MARGIN, y);
  y += 14;
  doc.text(`ID: ${exam.id}`, MARGIN, y);
  y += 20;
  doc.setTextColor(0);

  doc.setFontSize(12);
  const correctCount = exam.questions.filter((q) => q.isCorrect).length;
  doc.text(
    `Pontuação: ${exam.score}% · Acertos: ${correctCount} de ${exam.questions.length}`,
    MARGIN,
    y,
  );
  y += 16;
  doc.text(
    `Início: ${new Date(exam.createdAt).toLocaleString("pt-BR")}`,
    MARGIN,
    y,
  );
  y += 14;
  doc.text(
    `Fim: ${new Date(exam.finishedAt).toLocaleString("pt-BR")}`,
    MARGIN,
    y,
  );
  y += 24;

  for (let i = 0; i < exam.questions.length; i++) {
    const q = exam.questions[i];
    y = ensureSpace(doc, y, 60);
    doc.setFontSize(12);
    doc.setFont("helvetica", "bold");
    doc.text(`Questão ${i + 1}`, MARGIN, y);
    y += 16;
    doc.setFont("helvetica", "normal");

    const statementLines = doc.splitTextToSize(q.statement, CONTENT_WIDTH);
    y = ensureSpace(doc, y, statementLines.length * 14);
    doc.text(statementLines, MARGIN, y);
    y += statementLines.length * 14 + 6;

    for (const alt of q.alternatives) {
      const letter = letterFor(alt.position);
      const isCorrect = alt.id === q.correctAlternativeId;
      const isWrongPick =
        alt.id === q.selectedAlternativeId && !q.isCorrect;
      const marker = isCorrect ? "[✓]" : isWrongPick ? "[✗]" : "   ";
      const line = `${marker} ${letter}) ${alt.text}`;
      const wrapped = doc.splitTextToSize(line, CONTENT_WIDTH);
      y = ensureSpace(doc, y, wrapped.length * 14);
      doc.text(wrapped, MARGIN, y);
      y += wrapped.length * 14;
    }

    if (q.explanation) {
      y += 4;
      const wrapped = doc.splitTextToSize(
        `Explicação: ${q.explanation}`,
        CONTENT_WIDTH,
      );
      y = ensureSpace(doc, y, wrapped.length * 14);
      doc.setTextColor(80);
      doc.text(wrapped, MARGIN, y);
      doc.setTextColor(0);
      y += wrapped.length * 14;
    }

    y += 16;
  }

  return doc.output("blob");
}
